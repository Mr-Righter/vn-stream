cmake_minimum_required(VERSION 3.20)
project(onscripter_zero_linux LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET sdl2 SDL2_image SDL2_mixer SDL2_ttf freetype2 luajit fmt)
set(SOURCE "${ONS_SOURCE}")
if(NOT EXISTS "${SOURCE}/src/entry/onscripter_main.cpp")
    message(FATAL_ERROR "ONS_SOURCE must point to the pinned ONScripter-zero source")
endif()
file(GLOB SOURCES ${SOURCE}/src/*.cpp ${SOURCE}/src/charset/*.c ${SOURCE}/src/renderer/*.cpp ${SOURCE}/src/reader/*.cpp ${SOURCE}/src/onscripter/*.cpp ${SOURCE}/src/builtin_dll/*.cpp ${SOURCE}/src/language/*.cpp ${SOURCE}/src/private/*.cpp ${SOURCE}/src/resize/*.cpp)
list(REMOVE_ITEM SOURCES ${SOURCE}/src/AVIWrapper.cpp)
add_executable(onscripter ${SOURCE}/src/entry/onscripter_main.cpp ${SOURCE}/src/SDL2_rotozoom.c ${SOURCES})
target_include_directories(onscripter PRIVATE ${SOURCE}/src ${SOURCE}/src/onscripter ${SOURCE}/src/reader ${SOURCE}/include)
# Supply standard declarations missing from upstream transitive includes.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/portability.hpp" "#include <algorithm>\n#include <cstring>\n#include \"config.hpp\"\n")
target_precompile_headers(onscripter PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_BINARY_DIR}/portability.hpp>")
# Match upstream desktop features; __stdcall is empty for the Linux ABI.
target_compile_definitions(onscripter PRIVATE ONS_ZERO_VERSION="0.9.10-development" ONS_JH_VERSION="0.8.0" ONS_VERSION="20181218" NSC_VERSION=296 UTF8_FILESYSTEM=1 ONSCRIPTER_EXTEND_INIT=1 RENDER_COPY_RECT_FULL=1 USE_LUA=1 USE_BUILTIN_LAYER_EFFECTS=1 USE_BUILTIN_EFFECTS=1 USE_PARALLEL=1 FMT_HEADER_ONLY=1 USE_IMAGE_CACHE=1 USE_SIMD=1 __stdcall=)
target_compile_options(onscripter PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-mavx2>")
target_link_libraries(onscripter PRIVATE PkgConfig::DEPS jpeg bz2 pthread dl)
